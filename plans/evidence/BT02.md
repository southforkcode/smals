## Direct verification and reproduction

Create `temp/evidence/BT02/`, then write `good.cmake`:

```cmake
cmake_minimum_required(VERSION 3.28)
math(EXPR answer "6 * 7")
if(NOT answer EQUAL 42)
  message(FATAL_ERROR "Expected 42, received ${answer}")
endif()
message(STATUS "cmake smoke: ${answer}")
file(WRITE "${CMAKE_CURRENT_LIST_DIR}/good-result.txt" "${answer}\n")
```

Write `bad.cmake`:

```cmake
cmake_minimum_required(VERSION 3.28)
unknown_smoke_command()
file(WRITE "${CMAKE_CURRENT_LIST_DIR}/bad-result.txt" "unexpected\n")
```

Remove any prior `good-result.txt` and `bad-result.txt` from this temporary directory before running. From the repository root:

```sh
/opt/homebrew/Cellar/cmake/4.4.4/bin/cmake --version
/opt/homebrew/bin/brew list --versions cmake
/opt/homebrew/Cellar/cmake/4.4.4/bin/cmake -P temp/evidence/BT02/good.cmake
/opt/homebrew/Cellar/cmake/4.4.4/bin/cmake -P temp/evidence/BT02/bad.cmake
```
